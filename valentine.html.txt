<script>
    const noBtn = document.getElementById("noBtn");
    const yesBtn = document.getElementById("yesBtn");
    const questionScreen = document.getElementById("questionScreen");
    const celebrateScreen = document.getElementById("celebrateScreen");

    let x = Math.random() * (window.innerWidth - noBtn.offsetWidth);
    let y = Math.random() * (window.innerHeight - noBtn.offsetHeight);
    let dx = 2;
    let dy = 2;

    function moveNoButton() {
        x += dx;
        y += dy;

        if (x <= 0 || x >= window.innerWidth - noBtn.offsetWidth) {
            dx *= -1;
        }

        if (y <= 0 || y >= window.innerHeight - noBtn.offsetHeight) {
            dy *= -1;
        }

        noBtn.style.left = x + "px";
        noBtn.style.top = y + "px";
    }

    setInterval(moveNoButton, 16);

    yesBtn.addEventListener("click", () => {
        questionScreen.classList.add("hidden");
        celebrateScreen.classList.remove("hidden");
    });
</script>
